cmake_minimum_required(VERSION 3.16)
project(skia_d3d12_interop_fixture LANGUAGES C CXX)
include(CTest)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  add_library(d3d12-interop-fixture SHARED fixture.cpp)
  target_compile_definitions(d3d12-interop-fixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(d3d12-interop-fixture PRIVATE d3d12 dxgi dxguid)
  add_executable(interop-sdk-check sdk-check.c)
  add_test(NAME interop-windows-sdk-layouts COMMAND interop-sdk-check)
else()
  add_executable(interop-host-mirror sdk-check.c)
  add_test(NAME interop-host-mirror-not-windows COMMAND interop-host-mirror)
endif()
