cmake_minimum_required(VERSION 3.20)
project(skia_metal_interop_fixture LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "The Metal SDK fixture must be compiled with the Apple SDK")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET "10.15")
add_library(metal-interop-fixture SHARED fixture.mm)
target_compile_options(metal-interop-fixture PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(metal-interop-fixture PRIVATE "-framework Metal" "-framework Foundation")
add_executable(metal-interop-sdk-check sdk-check.mm)
target_compile_options(metal-interop-sdk-check PRIVATE -fobjc-arc -Wall -Wextra -Werror)
target_link_libraries(metal-interop-sdk-check PRIVATE "-framework Metal" "-framework Foundation")
enable_testing()
add_test(NAME metal-interop-sdk COMMAND metal-interop-sdk-check)
